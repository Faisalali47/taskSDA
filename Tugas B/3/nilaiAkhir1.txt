PROGRAM NilaiAkhir

DEKLARASI
    nama, nim : string
    nilaiFormatif, nilaiUTS, nilaiUAS : real
    nilaiAkhir : real
    nilaiHuruf : string

ALGORITMA
    read(nama)
    read(nim)
    read(nilaiFormatif)
    read(nilaiUTS)
    read(nilaiUAS)

    nilaiAkhir ← (nilaiFormatif * 30 / 100) +
                  (nilaiUTS * 30 / 100) +
                  (nilaiUAS * 40 / 100)

    if nilaiAkhir >= 80 then
        nilaiHuruf ← "A"
    else if nilaiAkhir >= 77 then
        nilaiHuruf ← "A-"
    else if nilaiAkhir >= 74 then
        nilaiHuruf ← "B+"
    else if nilaiAkhir >= 70 then
        nilaiHuruf ← "B"
    else if nilaiAkhir >= 67 then
        nilaiHuruf ← "B-"
    else if nilaiAkhir >= 64 then
        nilaiHuruf ← "C+"
    else if nilaiAkhir >= 60 then
        nilaiHuruf ← "C"
    else if nilaiAkhir >= 50 then
        nilaiHuruf ← "D"
    else
        nilaiHuruf ← "E"
    endif

    write(nama)
    write(nim)
    write(nilaiFormatif)
    write(nilaiUTS)
    write(nilaiUAS)
    write(nilaiAkhir)
    write(nilaiHuruf)
